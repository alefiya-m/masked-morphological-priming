# Unified Linear Mixed Effects Analysis 
# ==============================================================================
library(lme4)
library(lmerTest)
library(emmeans)

# Set up environment and read data
all.data <- read.csv("MPP_Data.csv")
colnames(all.data)[1] <- "Version"

# ------------------------------------------------------------------------------
# 1. Data Cleaning & Outlier Removal
# ------------------------------------------------------------------------------
clean.data <- subset(all.data, Acc_Word == 1 & RT_Word >= 200 & RT_Word <= 3000)

# Calculate subject-level Z-scores
subjmeans <- aggregate(RT_Word ~ Subject, data = clean.data, FUN = mean)
subjsd    <- aggregate(RT_Word ~ Subject, data = clean.data, FUN = sd)
names(subjmeans) <- c("Subject", "meanRT")
names(subjsd)    <- c("Subject", "sdRT")

clean1.data <- merge(clean.data, subjmeans, by = "Subject")
clean2.data <- merge(clean1.data, subjsd, by = "Subject")
clean2.data$ZRT <- (clean2.data$RT_Word - clean2.data$meanRT) / clean2.data$sdRT

# Retain valid word trials within 2.5 SDs
keep.data <- subset(clean2.data, ZRT >= -2.5 & ZRT <= 2.5 & Lexicality == "Word")

# ------------------------------------------------------------------------------
# 2. Mean-Centering Continuous Lexical Covariates
# ------------------------------------------------------------------------------
keep.data$Prime_Length_c      <- keep.data$Prime_Length - mean(keep.data$Prime_Length, na.rm = TRUE)
keep.data$Prime_Frequency_c   <- keep.data$Prime_Frequency - mean(keep.data$Prime_Frequency, na.rm = TRUE)
keep.data$Suffix_Length_c     <- keep.data$Suffix_Length - mean(keep.data$Suffix_Length, na.rm = TRUE)
keep.data$Target_Length_c     <- keep.data$Target_Length - mean(keep.data$Target_Length, na.rm = TRUE)
keep.data$Target_Frequency_c  <- keep.data$Target_Frequency - mean(keep.data$Target_Frequency, na.rm = TRUE)
keep.data$Target_OrthoN_c     <- keep.data$Target_OrthoN - mean(keep.data$Target_OrthoN, na.rm = TRUE)
keep.data$Target_Nsyll_c      <- keep.data$Target_Nsyll - mean(keep.data$Target_Nsyll, na.rm = TRUE)

# Ensure structural factors are evaluated correctly as categorical variables
keep.data$Relatedness <- as.factor(keep.data$Relatedness)
keep.data$Load        <- as.factor(keep.data$Load)
keep.data$PrimeType   <- as.factor(keep.data$PrimeType)

# ------------------------------------------------------------------------------
# 3. Setting Up Economical Contrast Coding (Fixed Matrix Syntax)
# ------------------------------------------------------------------------------

# Sum-code Relatedness: Unrelated (-0.5) vs Related (+0.5)
# Using dimnames to correctly assign the column label inside the matrix
contrasts(keep.data$Relatedness) <- matrix(c(-0.5, 0.5), 
                                           dimnames = list(NULL, "Priming_Effect"))

# Sum-code Load: Low (-0.5) vs High (+0.5)
contrasts(keep.data$Load)        <- matrix(c(-0.5, 0.5), 
                                           dimnames = list(NULL, "High_vs_Low"))

# Custom Helmert-style contrasts for PrimeType to isolate Transparent pairs
mat.PrimeType <- matrix(c(
  2/3, -1/3, -1/3,  # Transparent vs Average of Others
  0,    1/2, -1/2   # Opaque vs Form-Related
), nrow = 3, byrow = TRUE, 
dimnames = list(c("Transparent", "Opaque", "Form-Related"), c("Trans_vs_Others", "Opaque_vs_Form")))

# Apply the custom contrast matrix to PrimeType
contrasts(keep.data$PrimeType) <- mat.PrimeType

# ------------------------------------------------------------------------------
# 4. Unified Omnibus Mixed-Effects Model
# ------------------------------------------------------------------------------
# Including your continuous covariates alongside the complete 3-way experimental interaction.
# Random effects use random intercepts and uncorrelated random slopes to optimize convergence.
omnibus_model <- lmer(
  RT_Word ~ PrimeType * Relatedness * Load + 
            (1 + Relatedness + Load || Subject) + 
            (1 + Relatedness + Load || Target), 
  data = keep.data
)

print(summary(omnibus_model))

omnibus_model <- lmer(
  RT_Word ~ Prime_Length_c + Prime_Frequency_c + Suffix_Length_c + 
            Target_Length_c + Target_Frequency_c + Target_OrthoN_c + Target_Nsyll_c + 
            PrimeType * Relatedness * Load + (1 | Subject) + (1 | Target), data = keep.data)
            

# Run Type III ANOVA to formally test the PrimeType x Relatedness x Load interaction
print(anova(omnibus_model, type = "3"))

# ------------------------------------------------------------------------------
# 5. Targeted Simple Effects & Pairwise Comparisons via emmeans
# ------------------------------------------------------------------------------
cat("\n--- Estimated Marginal Means: Priming Effects Across Conditions ---\n")

# Extract the priming effect sizes (differences between Related and Unrelated)
priming_trends <- emtrends(omnibus_model, ~ PrimeType * Load, var = "Relatedness")
print(priming_trends)

cat("\n--- Testing If Load Moderates Priming Within Each Prime Type ---\n")
# Pairwise comparisons check if the priming effect differs between Low and High load for each Prime Type
load_moderation_tests <- pairs(priming_trends, by = "PrimeType")
print(load_moderation_tests)
2. Generating a Clean, Consistent Dataset
Because we want your structural contrasts to evaluate clean, predictable variances when testing the code script, you can generate an identical dummy verification dataset using the code below. It maps perfectly to your exact experimental design variables and creates a simulated data file (200825_Data.csv) inside your specified folder directory.

R
# ==============================================================================
# Simulation Script: Consistent Validation Dataset
# ==============================================================================
set.seed(42) # For reproducibility

# Define design parameters
n_subjects <- 40
n_targets  <- 120 # 40 items per PrimeType (Transparent, Opaque, Form-Related)

# Generate baseline data grid
subjects <- paste0("S", 1:n_subjects)
targets  <- paste0("T", 1:n_targets)

# Assign PrimeType to specific target items (across-items manipulation)
target_pool <- data.frame(
  Target = targets,
  PrimeType = rep(c("Transparent", "Opaque", "Form-Related"), each = n_targets/3),
  Prime_Length = round(runif(n_targets, 4, 9)),
  Prime_Frequency = round(rnorm(n_targets, 3, 1.2), 2),
  Suffix_Length = sample(c(2, 3, 4), n_targets, replace = TRUE),
  Target_Length = round(runif(n_targets, 4, 9)),
  Target_Frequency = round(rnorm(n_targets, 3.5, 1.1), 2),
  Target_OrthoN = round(runif(n_targets, 1, 15)),
  Target_Nsyll = sample(c(1, 2, 3), n_targets, replace = TRUE)
)

# Expand cross-product for within-subject factors
sim_data <- expand.grid(Subject = subjects, Target = targets)
sim_data <- merge(sim_data, target_pool, by = "Target")

# Randomly assign Relatedness and Load treatments within subjects
sim_data$Relatedness <- sample(c("Related", "Unrelated"), nrow(sim_data), replace = TRUE)
sim_data$Load        <- sample(c("Low", "High"), nrow(sim_data), replace = TRUE)
sim_data$Version     <- sample(c("V1", "V2", "V3"), nrow(sim_data), replace = TRUE)
sim_data$Acc_Word    <- sample(c(1, 1, 1, 1, 0), nrow(sim_data), replace = TRUE) # ~80% accuracy
sim_data$Lexicality  = "Word"

# Calculate underlying synthetic RTs to mirror your hypothesis:
# Priming occurs ONLY for Transparent pairs and is NOT affected by load.
base_rt <- 650
subject_rfx <- rnorm(n_subjects, 0, 40)[match(sim_data$Subject, subjects)]
target_rfx  <- rnorm(n_targets, 0, 30)[match(sim_data$Target, targets)]

sim_data$RT_Word <- base_rt + subject_rfx + target_rfx +
  (sim_data$Load == "High") * 45 +  # Main effect of load slowing down RT
  (sim_data$PrimeType == "Transparent" & sim_data$Relatedness == "Related") * -50 + # Transparent Priming
  rnorm(nrow(sim_data), 0, 60)      # Residual trial noise

# Trim RT range to match expected boundary criteria
sim_data$RT_Word <- pmax(200, pmin(3000, round(sim_data$RT_Word)))

# Save the dataset into your working directory
write.csv(sim_data, "MPP_results.csv", row.names = FALSE)
cat("Consistent verification dataset generated and saved successfully.\n")
